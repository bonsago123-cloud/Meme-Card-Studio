function render(c,w,h,preview=false){c.save();const free=state.canvasEnabled===false&&state.photo;const ox=free&&preview?state.photo.x:0,oy=free&&preview?state.photo.y:0;c.translate(ox,oy);c.clearRect(0,0,w,h);if((state.canvasEnabled!==false&&!state.photoCanvas)||$('format').value==='jpeg'){c.fillStyle=state.background;c.fillRect(0,0,w,h);}typography(c,w,h);if(image&&state.photo){let p=state.photo;c.drawImage(image,p.sx,p.sy,p.sw,p.sh,free?0:p.x,free?0:p.y,p.w,p.h);}c.save();c.textAlign='center';c.textBaseline='top';c.fillStyle=state.textColor;let size=state.fontSize,lines;const maxWidth=w*.88,maxHeight=h*.88;do{c.font=`700 ${size}px "Malgun Gothic", "Apple SD Gothic Neo", sans-serif`;lines=wrapText(state.text,maxWidth,t=>c.measureText(t).width);if(lines.length*size*1.35<=maxHeight)break;size-=1;}while(size>8);const lineH=size*1.35,bw=Math.min(maxWidth,Math.max(1,...lines.map(l=>c.measureText(l).width))),bh=state.text?lines.length*lineH:0;const x=clamp(state.textX/100*w,bw/2+12,w-bw/2-12),y=clamp(state.textY/100*h-bh/2,12,h-bh-12);lines.forEach((l,i)=>c.fillText(l,x,y+i*lineH));if(preview)textBox={x:x-bw/2+ox,y:y+oy,w:bw,h:bh};c.restore();c.restore();}
